# Reuse artifacts from the same bundled QuaZip/ZLIB build used by core.
# This does not discover or install another QuaZip package.
get_filename_component(BUNDLE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(QUAZIP_BUNDLE "${BUNDLE_DIR}/build/_deps")
if(EXISTS "${QUAZIP_BUNDLE}/zstd-build/lib/RelWithDebInfo/zstd_static.lib")
    add_executable(ColumnTileCacheTest ColumnTileCacheTest.cpp)
    target_compile_features(ColumnTileCacheTest PRIVATE cxx_std_17)
    target_include_directories(ColumnTileCacheTest PRIVATE "${QUAZIP_BUNDLE}/zstd-src/lib")
    target_link_libraries(ColumnTileCacheTest PRIVATE Qt6::Core "${QUAZIP_BUNDLE}/zstd-build/lib/RelWithDebInfo/zstd_static.lib")
    add_test(NAME ColumnTileCacheTest COMMAND ColumnTileCacheTest)
    add_executable(CacheCodecBenchmark CacheCodecBenchmark.cpp)
    target_compile_features(CacheCodecBenchmark PRIVATE cxx_std_17)
    target_include_directories(CacheCodecBenchmark PRIVATE "${QUAZIP_BUNDLE}/zstd-src/lib")
    target_link_libraries(CacheCodecBenchmark PRIVATE Qt6::Core "${QUAZIP_BUNDLE}/zstd-build/lib/RelWithDebInfo/zstd_static.lib")
endif()
if(EXISTS "${QUAZIP_BUNDLE}/quazip-build/quazip/RelWithDebInfo/quazip1-qt6.lib")
    add_executable(StoredZipRangesTest StoredZipRangesTest.cpp)
    target_compile_features(StoredZipRangesTest PRIVATE cxx_std_17)
    target_compile_definitions(StoredZipRangesTest PRIVATE QUAZIP_STATIC)

        "${QUAZIP_BUNDLE}/quazip-src" "${QUAZIP_BUNDLE}/quazip-build/quazip"
        "${QUAZIP_BUNDLE}/zlib-src" "${QUAZIP_BUNDLE}/zlib-build")
    target_link_libraries(StoredZipRangesTest PRIVATE Qt6::Core
        "${QUAZIP_BUNDLE}/quazip-build/quazip/RelWithDebInfo/quazip1-qt6.lib"
        "${QUAZIP_BUNDLE}/zlib-build/RelWithDebInfo/libzs.lib")
    add_test(NAME StoredZipRangesTest COMMAND StoredZipRangesTest)
    if(EXISTS "${QUAZIP_BUNDLE}/zstd-build/lib/RelWithDebInfo/zstd_static.lib")
